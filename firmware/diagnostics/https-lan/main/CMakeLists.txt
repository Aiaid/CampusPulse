function(cp_read_pem path output)
    file(READ "${path}" pem)
    if(NOT pem MATCHES "-----BEGIN CERTIFICATE-----" OR pem MATCHES "PRIVATE KEY")
        message(FATAL_ERROR "Diagnostic trust input must contain a public certificate only")
    endif()
    string(REPLACE "\\" "\\\\" pem "${pem}")
    string(REPLACE "\r" "" pem "${pem}")
    string(REPLACE "\"" "\\\"" pem "${pem}")
    string(REPLACE "\n" "\\n\"\n\"" pem "${pem}")
    set(${output} "${pem}" PARENT_SCOPE)
endfunction()
